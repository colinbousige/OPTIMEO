optimeo.doe

This module provides a class for creating and visualizing a design of experiments (DoE). It supports various types of designs including Full Factorial, Sobol sequence, Fractional Factorial, Definitive Screening Design, Space Filling Latin Hypercube, Randomized Latin Hypercube, Optimal, Plackett-Burman, and Box-Behnken. The class allows the user to specify the parameters, their types, and values, and generates the design accordingly. It also provides a method to plot the design using scatter plots.

 41    Class to create a design of experiments (DoE) for a given model.
 42    This class allows the user to specify the type of design, the parameters,

 47    Parameters
 48    ----------
 49    
 50    type : str
 51        The type of design to create. Must be one of:
 52        `'Full Factorial'`, `'Sobol sequence'`, `'Fractional Factorial'`,
 53        `'Definitive Screening'`, `'Space Filling Latin Hypercube'`,
 54        `'Randomized Latin Hypercube'`, `'Optimal'`, `'Plackett-Burman'`,
 55        `'Box-Behnken'` or `'Central Composite'`.
 56    parameters : List[Dict[str, Dict[str, Any]]]
 57        List of parameters for the design, each with a dictionary of properties.
 58        Each dictionary should contain 'name', 'type', and 'values'.
 59        'values' should be a list of possible values for the parameter.
 60        'type' should be either "int", "integer", "float", "<other>". 
 61        Any <other> will be considered as "categorical".
 62        'values' should be a list of possible values for the parameter.
 63    Nexp : int, optional
 64        Number of experiments in the design, when applicable. Default is 4.
 65    order : int, optional
 66        Order of the model (for 'Optimal' design). Default is 2.
 67    randomize : bool, optional
 68        Whether to randomize the run order. Default is True.

366    def create_design(self):
367        """
368        Create the design of experiments based on the specified type and parameters.
369        """
370        for par in self.parameters:
371            if par['type'].lower() == "categorical":
372                if self.type != 'Sobol sequence':
373                    le = LabelEncoder()
374                    label = le.fit_transform(par['values'])
375                    par['values'] = label
376                    par['encoder'] = le
377                    self.lows[par['name']] = np.min(par['values'])
378                    self.highs[par['name']] = np.max(par['values'])
379                else:
380                    par['encoder'] = None
381            else:
382                self.lows[par['name']] = np.min(par['values'])
383                self.highs[par['name']] = np.max(par['values'])
384
385        pars = {par['name']: par['values'] for par in self.parameters}
386
387        if self.type == 'Full Factorial':
388            self.design = build.full_fact(pars)
389        elif self.type == 'Sobol sequence':
390            from ax.adapter.registry import Generators
391            from ax.generation_strategy.generation_node import GenerationStep
392            from ax.generation_strategy.generation_strategy import GenerationStrategy
393            from ax.service.ax_client import AxClient, ObjectiveProperties
394
395            ax_client = AxClient()
396            params = []
397            for par in self.parameters:
398                if par['type'].lower() == "float":
399                    params.append({'name': par['name'],
400                                   'type': 'range',
401                                   'value_type': 'float',
402                                   'bounds': [float(np.min(par['values'])), float(np.max(par['values']))]})
403                elif par['type'].lower() in ["integer", 'int']:
404                    params.append({'name': par['name'],
405                                   'type': 'range',
406                                   'value_type': 'int',
407                                   'bounds': [int(np.min(par['values'])), int(np.max(par['values']))]})
408                else:
409                    params.append({'name': par['name'],
410                                   'type': 'choice',
411                                   'values': par['values']})
412            ax_client.create_experiment(
413                name="DOE",
414                parameters=params,
415                objectives={"response": ObjectiveProperties(minimize=False)},
416                parameter_constraints=self._feature_constraints
417            )
418            gs = GenerationStrategy(
419                steps=[GenerationStep(
420                    generator=Generators.SOBOL,
421                    num_trials=-1,
422                    should_deduplicate=True,
423                    model_kwargs={"seed": self.seed},
424                    model_gen_kwargs={},
425                )]
426            )
427            generated_runs = gs.gen(
428                experiment=ax_client.experiment,
429                data=None,
430                n=self.Nexp
431            )
432            generator_run = generated_runs[0][0]
433            if self.Nexp == 1:
434                ax_client.experiment.new_trial(generator_run)
435            else:
436                ax_client.experiment.new_batch_trial(generator_run)
437            trials = ax_client.get_trials_data_frame()
438            self.design = trials[trials['trial_status'] == 'CANDIDATE']
439            self.design = self._design.drop(columns=['trial_index',
440                                                      'trial_status',
441                                                      'arm_name',
442                                                      'generation_method',
443                                                      'generation_node'],
444                                           errors='ignore')
445        elif self.type == 'Fractional Factorial':
446            for par in range(len(self.parameters)):
447                if self.parameters[par]['type'] == "Numerical":
448                    self.parameters[par]['type'] = "Categorical"
449                    le = LabelEncoder()
450                    label = le.fit_transform(self.parameters[par]['values'])
451                    self.parameters[par]['values'] = label
452                    self.parameters[par]['encoder'] = le
453            design = gsd([len(par['values']) for par in self.parameters], self.reduction)
454            self.design = pd.DataFrame(design, columns=[par['name'] for par in self.parameters])
455        elif self.type == 'Definitive Screening':
456            params = {par['name']: [np.min(par['values']), np.max(par['values'])] for par in self.parameters}
457            self.design = dsd.generate(factors_dict=params)
458        elif self.type == 'Space Filling Latin Hypercube':
459            self.design = build.space_filling_lhs(pars, num_samples=self.Nexp)
460        elif self.type == 'Randomized Latin Hypercube':
461            self.design = build.lhs(pars, num_samples=self.Nexp)
462        elif self.type == 'Optimal':
463            reaction_design = build_optimal(
464                len(self.parameters),
465                order=ModelOrder(self.order),
466                run_count=self.Nexp)
467            reaction_design.columns = [par['name'] for par in self.parameters]
468            self.design = coded_to_actual(reaction_design, self._lows, self._highs)
469        elif self.type == 'Plackett-Burman':
470            self.design = build.plackett_burman(pars)
471        elif self.type == 'Box-Behnken':
472            if len(self.parameters) < 3 or any([len(par['values']) < 3 for par in self.parameters]):
473                self.design = pd.DataFrame({})
474                raise Warning("Box-Behnken design is not possible with less than 3 parameters and with less than 3 levels for any parameter.")
475            else:
476                self.design = build.box_behnken(d=pars)
477        elif self.type == 'Central Composite':
478            pars_list = {k: v.tolist() if hasattr(v, 'tolist') else list(v) for k, v in pars.items()}
479            self.design = build.central_composite(pars_list,
480                                                  center=self.center,
481                                                  alpha=self.alpha,
482                                                  face=self.face)
483        else:
484            raise ValueError("Unknown design type. Must be one of: 'Full Factorial', 'Sobol sequence', 'Fractional Factorial', 'Definitive Screening', 'Space Filling Latin Hypercube', 'Randomized Latin Hypercube', 'Optimal', 'Plackett-Burman', 'Box-Behnken' or 'Central Composite'.")
485
486        for par in self.parameters:
487            if par['type'].lower() == "categorical" and self.type != 'Sobol sequence':
488                vals = self._design[par['name']].to_numpy()
489                n_classes = len(par['encoder'].classes_)
490                clipped = [int(np.clip(np.round(v), 0, n_classes - 1)) for v in vals]
491                self.design[par['name']] = par['encoder'].inverse_transform(clipped)
492
493        # randomize the run order
494        self.design['run_order'] = np.arange(len(self._design)) + 1
495        if self.randomize:
496            ord = self._design['run_order'].to_numpy()
497            self.design['run_order'] = np.random.permutation(ord)
498        cols = self._design.columns.tolist()
499        cols = cols[-1:] + cols[:-1]
500        self.design = self._design[cols]
501        # apply the column types
502        for col in self._design.columns:
503            for par in self.parameters:
504                if col == par['name']:
505                    if par['type'].lower() == "float":
506                        self.design[col] = self._design[col].astype(float)
507                    elif par['type'].lower() in ["int", "integer"]:
508                        self.design[col] = self._design[col].astype(int)
509                    else:
510                        self.design[col] = self._design[col].astype(str)
511        return self._design

513    def plot(self):
514        """
515        Plot the design of experiments.
516
517        Returns
518        -------
519        List of plotly.graph_objs._figure.Figure
520            A list of Plotly figures representing the design of experiments.
521        """
522        fig = []
523        count = 0
524        if len(self.design) > 0:
525            if len(self.parameters) <= 2:
526                # Create 2D scatter plots
527                for i, faci in enumerate(self.parameters):
528                    for j, facj in enumerate(self.parameters):
529                        if j > i:
530                            fig.append(px.scatter(
531                                self.design,
532                                x=facj['name'],
533                                y=faci['name'],
534                                title=f"""{faci['name']} vs {facj['name']}""",
535                                labels={facj['name']: facj['name'], faci['name']: faci['name']}
536                            ))
537                            fig[count].update_traces(marker=dict(size=10))
538                            fig[count].update_layout(
539                                margin=dict(l=10, r=10, t=50, b=50),
540                                xaxis=dict(
541                                    showgrid=True,
542                                    gridcolor="lightgray",
543                                    zeroline=False,
544                                    showline=True,
545                                    linewidth=1,
546                                    linecolor="black",
547                                    mirror=True
548                                ),
549                                yaxis=dict(
550                                    showgrid=True,
551                                    gridcolor="lightgray",
552                                    zeroline=False,
553                                    showline=True,
554                                    linewidth=1,
555                                    linecolor="black",
556                                    mirror=True
557                                ),
558                            )
559                            count += 1
560            else:
561                # Create 3D scatter plots
562                for k, (faci, facj, fack) in enumerate(combinations(self.parameters, 3)):
563                    fig.append(go.Figure(data=[go.Scatter3d(
564                        x=self.design[facj['name']],
565                        y=self.design[faci['name']],
566                        z=self.design[fack['name']],
567                        mode='markers',
568                        marker=dict(size=10, color='royalblue', opacity=0.7),
569                    )]))
570                    fig[count].update_layout(
571                        template='ggplot2',
572                        height=500,
573                        width=500,
574                        scene=dict(
575                            xaxis_title=facj['name'],
576                            yaxis_title=faci['name'],
577                            zaxis_title=fack['name'],
578                        ),
579                        title=f"{faci['name']} vs {facj['name']}<br>vs {fack['name']}",
580                        margin=dict(l=10, r=10, t=50, b=50)
581                    )
582                    count += 1
583        return fig

367    def create_design(self):
368        """
369        Create the design of experiments based on the specified type and parameters.
370        """
371        for par in self.parameters:
372            if par['type'].lower() == "categorical":
373                if self.type != 'Sobol sequence':
374                    le = LabelEncoder()
375                    label = le.fit_transform(par['values'])
376                    par['values'] = label
377                    par['encoder'] = le
378                    self.lows[par['name']] = np.min(par['values'])
379                    self.highs[par['name']] = np.max(par['values'])
380                else:
381                    par['encoder'] = None
382            else:
383                self.lows[par['name']] = np.min(par['values'])
384                self.highs[par['name']] = np.max(par['values'])
385
386        pars = {par['name']: par['values'] for par in self.parameters}
387
388        if self.type == 'Full Factorial':
389            self.design = build.full_fact(pars)
390        elif self.type == 'Sobol sequence':
391            from ax.adapter.registry import Generators
392            from ax.generation_strategy.generation_node import GenerationStep
393            from ax.generation_strategy.generation_strategy import GenerationStrategy
394            from ax.service.ax_client import AxClient, ObjectiveProperties
395
396            ax_client = AxClient()
397            params = []
398            for par in self.parameters:
399                if par['type'].lower() == "float":
400                    params.append({'name': par['name'],
401                                   'type': 'range',
402                                   'value_type': 'float',
403                                   'bounds': [float(np.min(par['values'])), float(np.max(par['values']))]})
404                elif par['type'].lower() in ["integer", 'int']:
405                    params.append({'name': par['name'],
406                                   'type': 'range',
407                                   'value_type': 'int',
408                                   'bounds': [int(np.min(par['values'])), int(np.max(par['values']))]})
409                else:
410                    params.append({'name': par['name'],
411                                   'type': 'choice',
412                                   'values': par['values']})
413            ax_client.create_experiment(
414                name="DOE",
415                parameters=params,
416                objectives={"response": ObjectiveProperties(minimize=False)},
417                parameter_constraints=self._feature_constraints
418            )
419            gs = GenerationStrategy(
420                steps=[GenerationStep(
421                    generator=Generators.SOBOL,
422                    num_trials=-1,
423                    should_deduplicate=True,
424                    model_kwargs={"seed": self.seed},
425                    model_gen_kwargs={},
426                )]
427            )
428            generated_runs = gs.gen(
429                experiment=ax_client.experiment,
430                data=None,
431                n=self.Nexp
432            )
433            generator_run = generated_runs[0][0]
434            if self.Nexp == 1:
435                ax_client.experiment.new_trial(generator_run)
436            else:
437                ax_client.experiment.new_batch_trial(generator_run)
438            trials = ax_client.get_trials_data_frame()
439            self.design = trials[trials['trial_status'] == 'CANDIDATE']
440            self.design = self._design.drop(columns=['trial_index',
441                                                      'trial_status',
442                                                      'arm_name',
443                                                      'generation_method',
444                                                      'generation_node'],
445                                           errors='ignore')
446        elif self.type == 'Fractional Factorial':
447            for par in range(len(self.parameters)):
448                if self.parameters[par]['type'] == "Numerical":
449                    self.parameters[par]['type'] = "Categorical"
450                    le = LabelEncoder()
451                    label = le.fit_transform(self.parameters[par]['values'])
452                    self.parameters[par]['values'] = label
453                    self.parameters[par]['encoder'] = le
454            design = gsd([len(par['values']) for par in self.parameters], self.reduction)
455            self.design = pd.DataFrame(design, columns=[par['name'] for par in self.parameters])
456        elif self.type == 'Definitive Screening':
457            params = {par['name']: [np.min(par['values']), np.max(par['values'])] for par in self.parameters}
458            self.design = dsd.generate(factors_dict=params)
459        elif self.type == 'Space Filling Latin Hypercube':
460            self.design = build.space_filling_lhs(pars, num_samples=self.Nexp)
461        elif self.type == 'Randomized Latin Hypercube':
462            self.design = build.lhs(pars, num_samples=self.Nexp)
463        elif self.type == 'Optimal':
464            reaction_design = build_optimal(
465                len(self.parameters),
466                order=ModelOrder(self.order),
467                run_count=self.Nexp)
468            reaction_design.columns = [par['name'] for par in self.parameters]
469            self.design = coded_to_actual(reaction_design, self._lows, self._highs)
470        elif self.type == 'Plackett-Burman':
471            self.design = build.plackett_burman(pars)
472        elif self.type == 'Box-Behnken':
473            if len(self.parameters) < 3 or any([len(par['values']) < 3 for par in self.parameters]):
474                self.design = pd.DataFrame({})
475                raise Warning("Box-Behnken design is not possible with less than 3 parameters and with less than 3 levels for any parameter.")
476            else:
477                self.design = build.box_behnken(d=pars)
478        elif self.type == 'Central Composite':
479            pars_list = {k: v.tolist() if hasattr(v, 'tolist') else list(v) for k, v in pars.items()}
480            self.design = build.central_composite(pars_list,
481                                                  center=self.center,
482                                                  alpha=self.alpha,
483                                                  face=self.face)
484        else:
485            raise ValueError("Unknown design type. Must be one of: 'Full Factorial', 'Sobol sequence', 'Fractional Factorial', 'Definitive Screening', 'Space Filling Latin Hypercube', 'Randomized Latin Hypercube', 'Optimal', 'Plackett-Burman', 'Box-Behnken' or 'Central Composite'.")
486
487        for par in self.parameters:
488            if par['type'].lower() == "categorical" and self.type != 'Sobol sequence':
489                vals = self._design[par['name']].to_numpy()
490                n_classes = len(par['encoder'].classes_)
491                clipped = [int(np.clip(np.round(v), 0, n_classes - 1)) for v in vals]
492                self.design[par['name']] = par['encoder'].inverse_transform(clipped)
493
494        # randomize the run order
495        self.design['run_order'] = np.arange(len(self._design)) + 1
496        if self.randomize:
497            ord = self._design['run_order'].to_numpy()
498            self.design['run_order'] = np.random.permutation(ord)
499        cols = self._design.columns.tolist()
500        cols = cols[-1:] + cols[:-1]
501        self.design = self._design[cols]
502        # apply the column types
503        for col in self._design.columns:
504            for par in self.parameters:
505                if col == par['name']:
506                    if par['type'].lower() == "float":
507                        self.design[col] = self._design[col].astype(float)
508                    elif par['type'].lower() in ["int", "integer"]:
509                        self.design[col] = self._design[col].astype(int)
510                    else:
511                        self.design[col] = self._design[col].astype(str)
512        return self._design

371        for par in self.parameters:
372            if par['type'].lower() == "categorical":
373                if self.type != 'Sobol sequence':
374                    le = LabelEncoder()
375                    label = le.fit_transform(par['values'])
376                    par['values'] = label

378                    self.lows[par['name']] = np.min(par['values'])
379                    self.highs[par['name']] = np.max(par['values'])
380                else:
381                    par['encoder'] = None
382            else:
383                self.lows[par['name']] = np.min(par['values'])
384                self.highs[par['name']] = np.max(par['values'])
385
386        pars = {par['name']: par['values'] for par in self.parameters}
387
388        if self.type == 'Full Factorial':
389            self.design = build.full_fact(pars)
390        elif self.type == 'Sobol sequence':
391            from ax.adapter.registry import Generators
392            from ax.generation_strategy.generation_node import GenerationStep
393            from ax.generation_strategy.generation_strategy import GenerationStrategy
394            from ax.service.ax_client import AxClient, ObjectiveProperties
395
396            ax_client = AxClient()
397            params = []
398            for par in self.parameters:
399                if par['type'].lower() == "float":
400                    params.append({'name': par['name'],
401                                   'type': 'range',
402                                   'value_type': 'float',
403                                   'bounds': [float(np.min(par['values'])), float(np.max(par['values']))]})
404                elif par['type'].lower() in ["integer", 'int']:
405                    params.append({'name': par['name'],
406                                   'type': 'range',
407                                   'value_type': 'int',
408                                   'bounds': [int(np.min(par['values'])), int(np.max(par['values']))]})
409                else:
410                    params.append({'name': par['name'],
411                                   'type': 'choice',
412                                   'values': par['values']})
413            ax_client.create_experiment(
414                name="DOE",
415                parameters=params,
416                objectives={"response": ObjectiveProperties(minimize=False)},
417                parameter_constraints=self._feature_constraints
418            )
419            gs = GenerationStrategy(
420                steps=[GenerationStep(
421                    generator=Generators.SOBOL,
422                    num_trials=-1,
423                    should_deduplicate=True,
424                    model_kwargs={"seed": self.seed},
425                    model_gen_kwargs={},
426                )]
427            )
428            generated_runs = gs.gen(
429                experiment=ax_client.experiment,
430                data=None,
431                n=self.Nexp
432            )
433            generator_run = generated_runs[0][0]
434            if self.Nexp == 1:
435                ax_client.experiment.new_trial(generator_run)
436            else:
437                ax_client.experiment.new_batch_trial(generator_run)
438            trials = ax_client.get_trials_data_frame()
439            self.design = trials[trials['trial_status'] == 'CANDIDATE']
440            self.design = self._design.drop(columns=['trial_index',
441                                                      'trial_status',
442                                                      'arm_name',
443                                                      'generation_method',
444                                                      'generation_node'],
445                                           errors='ignore')
446        elif self.type == 'Fractional Factorial':
447            for par in range(len(self.parameters)):
448                if self.parameters[par]['type'] == "Numerical":
449                    self.parameters[par]['type'] = "Categorical"
450                    le = LabelEncoder()
451                    label = le.fit_transform(self.parameters[par]['values'])
452                    self.parameters[par]['values'] = label
453                    self.parameters[par]['encoder'] = le
454            design = gsd([len(par['values']) for par in self.parameters], self.reduction)
455            self.design = pd.DataFrame(design, columns=[par['name'] for par in self.parameters])
456        elif self.type == 'Definitive Screening':
457            params = {par['name']: [np.min(par['values']), np.max(par['values'])] for par in self.parameters}
458            self.design = dsd.generate(factors_dict=params)
459        elif self.type == 'Space Filling Latin Hypercube':
460            self.design = build.space_filling_lhs(pars, num_samples=self.Nexp)
461        elif self.type == 'Randomized Latin Hypercube':
462            self.design = build.lhs(pars, num_samples=self.Nexp)
463        elif self.type == 'Optimal':
464            reaction_design = build_optimal(
465                len(self.parameters),
466                order=ModelOrder(self.order),
467                run_count=self.Nexp)
468            reaction_design.columns = [par['name'] for par in self.parameters]
469            self.design = coded_to_actual(reaction_design, self._lows, self._highs)
470        elif self.type == 'Plackett-Burman':
471            self.design = build.plackett_burman(pars)
472        elif self.type == 'Box-Behnken':
473            if len(self.parameters) < 3 or any([len(par['values']) < 3 for par in self.parameters]):
474                self.design = pd.DataFrame({})
475                raise Warning("Box-Behnken design is not possible with less than 3 parameters and with less than 3 levels for any parameter.")
476            else:
477                self.design = build.box_behnken(d=pars)
478        elif self.type == 'Central Composite':
479            pars_list = {k: v.tolist() if hasattr(v, 'tolist') else list(v) for k, v in pars.items()}
480            self.design = build.central_composite(pars_list,
481                                                  center=self.center,
482                                                  alpha=self.alpha,
483                                                  face=self.face)
484        else:
485            raise ValueError("Unknown design type. Must be one of: 'Full Factorial', 'Sobol sequence', 'Fractional Factorial', 'Definitive Screening', 'Space Filling Latin Hypercube', 'Randomized Latin Hypercube', 'Optimal', 'Plackett-Burman', 'Box-Behnken' or 'Central Composite'.")
486
487        for par in self.parameters:
488            if par['type'].lower() == "categorical" and self.type != 'Sobol sequence':
489                vals = self._design[par['name']].to_numpy()
490                n_classes = len(par['encoder'].classes_)
491                clipped = [int(np.clip(np.round(v), 0, n_classes - 1)) for v in vals]
492                self.design[par['name']] = par['encoder'].inverse_transform(clipped)
493
494        # randomize the run order
495        self.design['run_order'] = np.arange(len(self._design)) + 1
496        if self.randomize:
497            ord = self._design['run_order'].to_numpy()
498            self.design['run_order'] = np.random.permutation(ord)
499        cols = self._design.columns.tolist()
500        cols = cols[-1:] + cols[:-1]
501        self.design = self._design[cols]
502        # apply the column types
503        for col in self._design.columns:
504            for par in self.parameters:
505                if col == par['name']:
506                    if par['type'].lower() == "float":
507                        self.design[col] = self._design[col].astype(float)
508                    elif par['type'].lower() in ["int", "integer"]:
509                        self.design[col] = self._design[col].astype(int)
510                    else:
511                        self.design[col] = self._design[col].astype(str)
512        return self._design
